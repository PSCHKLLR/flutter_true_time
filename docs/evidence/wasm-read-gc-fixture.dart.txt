import 'dart:js_interop';

import 'package:flutter_true_time/core.dart';

@JS('globalThis.trueTimeBench')
external set _bench(JSFunction value);

@JS('globalThis.trueTimeClockDomain')
external set _clockDomain(JSString value);

@JS('globalThis.trueTimeAllocationControl')
external set _allocationControl(JSFunction value);

class _AllocationProbe {
  _AllocationProbe(this.value);
  final int value;
}

List<_AllocationProbe>? _retained;

class _FixtureSource implements TimeSource {
  const _FixtureSource(this.id);

  @override
  final String id;
  @override
  String get endpoint => id;
  @override
  String get providerGroup => id;
  @override
  String get configurationKey => id;

  @override
  Future<TimeSample> query(
    MonotonicClock clock,
    Duration timeout, {
    CancellationSignal? cancellation,
  }) async {
    final at = clock.snapshot();
    return TimeSample(
      utc: DateTime.utc(2026, 11, 11).add(at.elapsed),
      at: at,
      uncertainty: const Duration(milliseconds: 10),
    );
  }
}

Future<void> main() async {
  final engine = TrustedTimeEngine(
    sources: const [_FixtureSource('a'), _FixtureSource('b')],
    autoRefresh: false,
  );
  if (!(await engine.sync()).success) throw StateError('Fixture sync failed');
  _clockDomain = engine.clock.capabilities.domain.toJS;
  _allocationControl = ((JSNumber count) {
    final previous = _retained?.last.value ?? 0;
    final retained = List.generate(count.toDartInt, _AllocationProbe.new);
    _retained = retained;
    return (previous ^ retained.last.value).toJS;
  }).toJS;
  _bench = ((JSBoolean detailed, JSNumber count) {
    final includeMetadata = detailed.toDart;
    final iterations = count.toDartInt;
    var checksum = 0;
    for (var i = 0; i < iterations; i++) {
      if (includeMetadata) {
        final reading = engine.read();
        checksum ^= reading.utc.microsecondsSinceEpoch ^ reading.uncertainty.inMicroseconds;
      } else {
        checksum ^= engine.now().microsecondsSinceEpoch;
      }
    }
    return checksum.toJS;
  }).toJS;
}
