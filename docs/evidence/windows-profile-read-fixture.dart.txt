import 'dart:developer';
import 'dart:io';

import 'package:flutter_true_time/core.dart';

class _FixtureSource implements TimeSource {
  const _FixtureSource(this.id);

  @override
  final String id;
  @override
  String get endpoint => id;
  @override
  String get providerGroup => id;
  @override
  String get configurationKey => id;

  @override
  Future<TimeSample> query(
    MonotonicClock clock,
    Duration timeout, {
    CancellationSignal? cancellation,
  }) async {
    final at = clock.snapshot();
    return TimeSample(
      utc: DateTime.utc(2026, 11, 11).add(at.elapsed),
      at: at,
      uncertainty: const Duration(milliseconds: 10),
    );
  }
}

final _retained = <_AllocationMarker>[];

final class _AllocationMarker {
  _AllocationMarker(this.value);
  final int value;
}

Future<void> main(List<String> args) async {
  final engine = TrustedTimeEngine(
    sources: const [_FixtureSource('a'), _FixtureSource('b')],
    autoRefresh: false,
  );
  if (!(await engine.sync()).success) throw StateError('Fixture sync failed');
  registerExtension('ext.true_time.measure', (_, params) async {
    final mode = params['mode'];
    final calls = int.parse(params['calls']!);
    _retained.clear();
    final int Function(int) sample = switch (mode) {
      'baseline' => (i) => i,
      'control' => (i) {
        final marker = _AllocationMarker(i);
        _retained.add(marker);
        return marker.value;
      },
      'now' => (_) => engine.now().microsecondsSinceEpoch,
      'read' => (_) {
        final value = engine.read();
        return value.utc.microsecondsSinceEpoch ^ value.uncertainty.inMicroseconds;
      },
      _ => throw ArgumentError.value(mode),
    };
    var checksum = 0;
    for (var i = 0; i < calls; i++) {
      checksum ^= sample(i);
    }
    return ServiceExtensionResponse.result('{"checksum":$checksum}');
  });
  final info = await Service.getInfo();
  File(args.single).writeAsStringSync(info.serverUri.toString());
}
