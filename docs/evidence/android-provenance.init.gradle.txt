import groovy.json.JsonOutput
import java.security.MessageDigest
import org.gradle.api.artifacts.component.ModuleComponentIdentifier

gradle.projectsEvaluated {
    def app = gradle.rootProject.findProject(':app')
    if (app == null || !app.plugins.hasPlugin('com.android.application')) return
    app.tasks.register('firstReleaseRuntimeInventory') {
        doLast {
            def runtime = app.configurations.getByName('releaseRuntimeClasspath')
            def modules = runtime.incoming.resolutionResult.allComponents.collect { it.id }
                .findAll { it instanceof ModuleComponentIdentifier }
                .collect { "${it.group}:${it.module}:${it.version}" }.sort().unique()
            def artifacts = runtime.incoming.artifactView {
                componentFilter { it instanceof ModuleComponentIdentifier }
            }.artifacts.artifacts.collect { artifact ->
                def id = artifact.id.componentIdentifier
                def hash = MessageDigest.getInstance('SHA-256')
                artifact.file.withInputStream { stream ->
                    byte[] buffer = new byte[65536]
                    for (int length; (length = stream.read(buffer)) != -1;) {
                        hash.update(buffer, 0, length)
                    }
                }
                [module: "${id.group}:${id.module}:${id.version}", file: artifact.file.name,
                 sha256: hash.digest().encodeHex().toString()]
            }.sort { "${it.module}:${it.file}" }
            println('FIRST_RELEASE_INVENTORY=' + JsonOutput.toJson([modules: modules, artifacts: artifacts]))
        }
    }
}
